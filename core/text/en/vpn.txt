[vpn.usage]
Usage: liltress vpn <command>

  status
  list
  import wireguard <name> <file>
  import openvpn <name> <file>
  up <name>
  down
  switch <name>
  boot <name>
  boot off
  rm <name>
  bypass
  bypass reload
  refresh-endpoint <name>
