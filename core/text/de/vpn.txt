[vpn.usage]
Aufruf: liltress vpn <Befehl>

  status
  list
  import wireguard <Name> <Datei>
  import openvpn <Name> <Datei>
  test <Name>
  switch <Name>
  mode on
  mode off
  rm <Name>
  bypass
  bypass reload
  refresh-endpoint <Name>
