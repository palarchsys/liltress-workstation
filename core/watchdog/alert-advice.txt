[aide.kernel-boot]

command<<

sudo bash << 'EOF'
for p in {paths}; do
  printf '%s\n' "=== $p"
  stat -c '%n %a %U:%G %y' -- "$p" || true
  dpkg -S -- "$p" || echo "HORS PAQUET $p"
  pkg="$(dpkg -S -- "$p" 2>/dev/null | awk -F: 'NR==1 { print $1 }' | awk -F, '{ print $1 }')"
  if [[ -n "$pkg" ]]; then
    dpkg -V "$pkg" 2>/dev/null | grep -F -- "$p" || echo "dpkg -V : aucune ligne pour ce chemin"
  fi
  grep -F -- "$p" /var/log/apt/history.log /var/log/dpkg.log 2>/dev/null || echo "journal apt/dpkg : aucune ligne pour ce chemin"
done
EOF
command>>

[aide.preload]
tool=aide
hold=1
notice=alert.aide.preload.notice
title=alert.aide.preload.title
match=/etc/ld\.so\.preload\b
command<<
sudo bash << 'EOF'
if [[ -f /etc/ld.so.preload ]]; then
  echo "présent"
  cat /etc/ld.so.preload
  while IFS= read -r so || [[ -n "$so" ]]; do
    case "$so" in
      ""|#*) continue ;;
    esac
    stat -c '%n %a %U:%G' -- "$so" || true
    dpkg -S -- "$so" || echo "HORS PAQUET $so"
  done < /etc/ld.so.preload
else
  echo "absent"
fi
EOF
command>>
