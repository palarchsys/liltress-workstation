[vpn.usage]
Usage : liltress vpn <commande>

  status
  list
  import openvpn <marque> <nom> <archive ou dossier>
  import wireguard <marque> <nom> <archive ou dossier>
  test <nom>
  switch <nom>
  mode on
  mode off
  rm <nom>
  bypass
  bypass reload
  refresh-endpoint <nom>
